 567    /// <summary>
 568    /// Claim this path (pooling).
 569    /// A claim on a path will ensure that it is not pooled.
 570    /// If you are using a path, you will want to claim it when you first get it and then release it when you will not
 571    /// use it anymore. When there are no claims on the path, it will be reset and put in a pool.
 572    ///
 573    /// This is essentially just reference counting.
 574    ///
 575    /// The object passed to this method is merely used as a way to more easily detect when pooling is not done correctl
 576    /// It can be any object, when used from a movement script you can just pass "this". This class will throw an except
 577    /// if you try to call Claim on the same path twice with the same object (which is usually not what you want) or
 578    /// if you try to call Release with an object that has not been used in a Claim call for that path.
 579    /// The object passed to the Claim method needs to be the same as the one you pass to this method.
 580    ///
 581    /// See: Release
 582    /// See: Pool
 583    /// See: pooling (view in online documentation for working links)
 584    /// See: https://en.wikipedia.org/wiki/Reference_counting
 585    /// </summary>
 586    public void Claim (System.Object o) {
 587      if (System.Object.ReferenceEquals(o, null)) throw new System.ArgumentNullException("o");
 588
 589      for (int i = 0; i < claimed.Count; i++) {
 590        // Need to use ReferenceEquals because it might be called from another thread
 591        if (System.Object.ReferenceEquals(claimed[i], o))
 592          throw new System.ArgumentException("You have already claimed the path with that object ("+o+"). Are you claimi
 593      }
 594
 595      claimed.Add(o);
 596#if ASTAR_POOL_DEBUG
 597      claimInfo.Add(o.ToString() + "\n\nClaimed from:\n" + System.Environment.StackTrace);
 598#endif
 599    }

 610    /// <summary>
 611    /// Releases a path claim (pooling).
 612    /// Removes the claim of the path by the specified object.
 613    /// When the claim count reaches zero, the path will be pooled, all variables will be cleared and the path will be p
 614    /// This is great for performance since fewer allocations are made.
 615    ///
 616    /// If the silent parameter is true, this method will remove the claim by the specified object
 617    /// but the path will not be pooled if the claim count reches zero unless a Release call (not silent) has been made 
 618    /// This is used by the internal pathfinding components such as Seeker and AstarPath so that they will not cause pat
 619    /// This enables users to skip the claim/release calls if they want without the path being pooled by the Seeker or A
 620    /// thus causing strange bugs.
 621    ///
 622    /// See: Claim
 623    /// See: PathPool
 624    /// </summary>
 625    public void Release (System.Object o, bool silent = false) {
 626      if (o == null) throw new System.ArgumentNullException("o");
 627
 628      for (int i = 0; i < claimed.Count; i++) {
 629        // Need to use ReferenceEquals because it might be called from another thread
 630        if (System.Object.ReferenceEquals(claimed[i], o)) {
 631          claimed.RemoveAt(i);
 632#if ASTAR_POOL_DEBUG
 633          claimInfo.RemoveAt(i);
 634#endif
 635          if (!silent) {
 636            releasedNotSilent = true;
 637          }
 638
 639          if (claimed.Count == 0 && releasedNotSilent) {
 640            PathPool.Pool(this);
 641          }
 642          return;
 643        }
 644      }
 645      if (claimed.Count == 0) {
 646        throw new System.ArgumentException("You are releasing a path which is not claimed at all (most likely it has bee
 647          "Are you releasing the path with the same object ("+o+") twice?" +
 648          "\nCheck out the documentation on path pooling for help.");
 649      }
 650      throw new System.ArgumentException("You are releasing a path which has not been claimed with this object ("+o+"). 
 651        "Are you releasing the path with the same object twice?\n" +
 652        "Check out the documentation on path pooling for help.");
 653    }

 655    /// <summary>
 656    /// Traces the calculated path from the end node to the start.
 657    /// This will build an array (<see cref="path)"/> of the nodes this path will pass through and also set the <see cre
 658    /// Assumes the <see cref="vectorPath"/> and <see cref="path"/> are empty and not null (which will be the case for a
 659    /// </summary>
 660    protected virtual void Trace (PathNode from) {
 661      // Current node we are processing
 662      PathNode c = from;
 663      int count = 0;
 664
 665      while (c != null) {
 666        c = c.parent;
 667        count++;
 668        if (count > 2048) {
 669          Debug.LogWarning("Infinite loop? >2048 node path. Remove this message if you really have that long paths (Path
 670          break;
 671        }
 672      }
 673
 674      // Ensure capacities for lists
 675      AstarProfiler.StartProfile("Check List Capacities");
 676
 677      if (path.Capacity < count) path.Capacity = count;
 678      if (vectorPath.Capacity < count) vectorPath.Capacity = count;
 679
 680      AstarProfiler.EndProfile();
 681
 682      c = from;
 683
 684      for (int i = 0; i < count; i++) {
 685        path.Add(c.node);
 686        c = c.parent;
 687      }
 688
 689      // Reverse
 690      int half = count/2;
 691      for (int i = 0; i < half; i++) {
 692        var tmp = path[i];
 693        path[i] = path[count-i-1];
 694        path[count - i - 1] = tmp;
 695      }
 696
 697      for (int i = 0; i < count; i++) {
 698        vectorPath.Add((Vector3)path[i].position);
 699      }
 700    }
