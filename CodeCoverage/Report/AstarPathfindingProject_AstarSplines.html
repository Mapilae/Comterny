 454    public static bool IsColinear (Vector3 a, Vector3 b, Vector3 c) {
 455      var lhs = b - a;
 456      var rhs = c - a;
 457      // Take the cross product of lhs and rhs
 458      // The magnitude of the cross product will be zero if the points a,b,c are colinear
 459      float x = lhs.y * rhs.z - lhs.z * rhs.y;
 460      float y = lhs.z * rhs.x - lhs.x * rhs.z;
 461      float z = lhs.x * rhs.y - lhs.y * rhs.x;
 462      float v = x*x + y*y + z*z;
 463
 464      // Epsilon not chosen with much thought, just that float.Epsilon was a bit too small.
 465      return v <= 0.0000001f;
 466    }
